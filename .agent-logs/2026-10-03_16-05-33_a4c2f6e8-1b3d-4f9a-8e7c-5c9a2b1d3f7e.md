import { z } from 'zod';

// ============================================================================
// BASE SCHEMAS - Common types used across contracts
// ============================================================================

const UUIDSchema = z.string().uuid('Invalid UUID format');
const EmailSchema = z.string().email('Invalid email format');
const DateTimeSchema = z.coerce.date();

const PaginationSchema = z.object({
  page: z.number().int().positive().default(1),
  limit: z.number().int().min(1).max(100).default(20),
  total: z.number().int().nonnegative(),
  has_more: z.boolean(),
});

export type Pagination = z.infer<typeof PaginationSchema>;

// ============================================================================
// ASSET FILE METADATA SCHEMAS
// ============================================================================

const AssetTypeEnum = z.enum(['image', 'video', 'audio', 'document']);

const FileDimensionsSchema = z.object({
  width: z.number().int().positive().optional(),
  height: z.number().int().positive().optional(),
  duration_ms: z.number().int().positive().optional(),
}).strict();

export type FileDimensions = z.infer<typeof FileDimensionsSchema>;

// ============================================================================
// REQUEST SCHEMAS
// ============================================================================

// POST /api/assets/upload - Request validation
export const AssetUploadRequestSchema = z.object({
  file: z.instanceof(File)
    .refine((file) => file.size > 0, 'File cannot be empty')
    .refine((file) => file.size <= 5 * 1024 * 1024, 'File must be < 5MB')
    .refine(
      (file) => ['image/png', 'image/jpeg', 'video/mp4', 'audio/mpeg', 'application/pdf']
        .includes(file.type),
      'Unsupported file type'
    ),
  name: z.string()
    .min(1, 'Asset name required')
    .max(255, 'Asset name max 255 chars')
    .regex(/^[a-zA-Z0-9._\- ]+$/, 'Invalid characters in name'),
  description: z.string().max(1000).optional(),
  asset_type: AssetTypeEnum,
});

export type AssetUploadRequest = z.infer<typeof AssetUploadRequestSchema>;

// GET /api/assets - Query parameters
export const AssetListQuerySchema = z.object({
  workspace_id: UUIDSchema,
  asset_type: AssetTypeEnum.optional(),
  created_after: DateTimeSchema.optional(),
  created_before: DateTimeSchema.optional(),
  sort_by: z.enum(['created_at', 'name', 'file_size']).default('created_at'),
  sort_order: z.enum(['asc', 'desc']).default('desc'),
  page: z.number().int().positive().default(1),
  limit: z.number().int().min(1).max(100).default(20),
});

export type AssetListQuery = z.infer<typeof AssetListQuerySchema>;

// PATCH /api/assets/:id/metadata - Metadata update
export const AssetMetadataUpdateSchema = z.object({
  metadata: z.record(z.unknown())
    .refine((obj) => Object.keys(obj).length <= 50, 'Max 50 metadata fields')
    .refine((obj) => JSON.stringify(obj).length <= 10000, 'Metadata too large'),
});

export type AssetMetadataUpdate = z.infer<typeof AssetMetadataUpdateSchema>;

// ============================================================================
// RESPONSE SCHEMAS - Public (no sensitive data)
// ============================================================================

// Asset public response (safe to return to client)
export const AssetPublicSchema = z.object({
  id: UUIDSchema,
  asset_type: AssetTypeEnum,
  name: z.string(),
  description: z.string().nullable(),
  file_size: z.number().int().positive(),
  mime_type: z.string(),
  dimensions: FileDimensionsSchema.nullable(),
  created_at: DateTimeSchema,
  updated_at: DateTimeSchema,
  // Note: workspace_id and created_by intentionally excluded for public API
}).strict();

export type AssetPublic = z.infer<typeof AssetPublicSchema>;

// Asset with metadata response
export const AssetWithMetadataSchema = AssetPublicSchema.extend({
  metadata: z.record(z.unknown()),
});

export type AssetWithMetadata = z.infer<typeof AssetWithMetadataSchema>;

// List response
export const AssetListResponseSchema = z.object({
  data: z.array(AssetPublicSchema),
  pagination: PaginationSchema,
});

export type AssetListResponse = z.infer<typeof AssetListResponseSchema>;

// Upload response
export const AssetUploadResponseSchema = z.object({
  id: UUIDSchema,
  asset_type: AssetTypeEnum,
  name: z.string(),
  file_path: z.string(), // Relative path like "/uploads/abc123/file.png"
  file_size: z.number(),
  created_at: DateTimeSchema,
});

export type AssetUploadResponse = z.infer<typeof AssetUploadResponseSchema>;

// ============================================================================
// ERROR RESPONSE SCHEMAS
// ============================================================================

export const ErrorResponseSchema = z.object({
  error: z.string(),
  message: z.string(),
  status: z.number().int(),
  timestamp: DateTimeSchema,
});

export const ValidationErrorSchema = z.object({
  error: z.literal('VALIDATION_ERROR'),
  message: z.string(),
  fields: z.record(z.array(z.string())),
  status: z.literal(400),
  timestamp: DateTimeSchema,
});

export type ValidationError = z.infer<typeof ValidationErrorSchema>;

// ============================================================================
// HELPER FUNCTIONS
// ============================================================================

export function parseAssetUploadRequest(data: unknown) {
  return AssetUploadRequestSchema.parse(data);
}

export function parseAssetListQuery(data: unknown) {
  return AssetListQuerySchema.parse(data);
}

export function createAssetResponse(asset: any): AssetPublic {
  return AssetPublicSchema.parse({
    id: asset.id,
    asset_type: asset.asset_type,
    name: asset.name,
    description: asset.description,
    file_size: asset.file_size,
    mime_type: asset.mime_type,
    dimensions: asset.dimensions,
    created_at: asset.created_at,
    updated_at: asset.updated_at,
    // workspace_id and created_by intentionally not included
  });
}

export function validateMetadataUpdate(data: unknown) {
  return AssetMetadataUpdateSchema.parse(data);
}
