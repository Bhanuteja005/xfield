'use client';

import React, { useReducer, useCallback, useMemo, useEffect } from 'react';
import { FixedSizeGrid as Grid } from 'react-window';
import AutoSizer from 'react-virtualized-auto-sizer';
import { AssetPublic } from '@xfield/shared/types';
import { AssetFilter } from './AssetFilter';
import { AssetSearch } from './AssetSearch';
import { AssetTile } from './AssetTile';
import { useAssetGallery } from './hooks/useAssetGallery';

interface AssetGalleryProps {
  workspaceId: string;
  onAssetSelect?: (asset: AssetPublic) => void;
  emptyState?: React.ReactNode;
}

interface GalleryState {
  selectedType: string | null;
  searchQuery: string;
  currentPage: number;
  itemsPerPage: number;
}

type GalleryAction =
  | { type: 'SET_FILTER'; payload: string | null }
  | { type: 'SET_SEARCH'; payload: string }
  | { type: 'SET_PAGE'; payload: number }
  | { type: 'RESET' };

const initialState: GalleryState = {
  selectedType: null,
  searchQuery: '',
  currentPage: 1,
  itemsPerPage: 20,
};

function galleryReducer(state: GalleryState, action: GalleryAction): GalleryState {
  switch (action.type) {
    case 'SET_FILTER':
      return { ...state, selectedType: action.payload, currentPage: 1 };
    case 'SET_SEARCH':
      return { ...state, searchQuery: action.payload, currentPage: 1 };
    case 'SET_PAGE':
      return { ...state, currentPage: action.payload };
    case 'RESET':
      return initialState;
    default:
      return state;
  }
}

export const AssetGallery: React.FC<AssetGalleryProps> = ({
  workspaceId,
  onAssetSelect,
  emptyState,
}) => {
  const [state, dispatch] = useReducer(galleryReducer, initialState);
  const { assets, isLoading, error, total } = useAssetGallery(workspaceId);

  // Filter assets based on current state
  const filteredAssets = useMemo(() => {
    let result = [...assets];

    // Filter by type
    if (state.selectedType) {
      result = result.filter((a) => a.asset_type === state.selectedType);
    }

    // Filter by search query
    if (state.searchQuery) {
      const query = state.searchQuery.toLowerCase();
      result = result.filter(
        (a) =>
          a.name.toLowerCase().includes(query) ||
          a.description?.toLowerCase().includes(query)
      );
    }

    return result;
  }, [assets, state.selectedType, state.searchQuery]);

  // Pagination
  const paginatedAssets = useMemo(() => {
    const start = (state.currentPage - 1) * state.itemsPerPage;
    const end = start + state.itemsPerPage;
    return filteredAssets.slice(start, end);
  }, [filteredAssets, state.currentPage, state.itemsPerPage]);

  const totalPages = Math.ceil(filteredAssets.length / state.itemsPerPage);

  const handleFilterChange = useCallback((type: string | null) => {
    dispatch({ type: 'SET_FILTER', payload: type });
  }, []);

  const handleSearch = useCallback((query: string) => {
    dispatch({ type: 'SET_SEARCH', payload: query });
  }, []);

  const handlePageChange = useCallback((page: number) => {
    dispatch({ type: 'SET_PAGE', payload: page });
  }, []);

  if (error) {
    return (
      <div
        className="p-6 bg-red-50 border border-red-200 rounded-lg"
        role="alert"
      >
        <h2 className="text-red-900 font-semibold">Failed to load assets</h2>
        <p className="text-red-700 text-sm">{error.message}</p>
      </div>
    );
  }

  return (
    <div className="space-y-4">
      {/* Header with search and filters */}
      <div className="bg-white rounded-lg shadow-sm p-4 space-y-4">
        <h2 className="text-lg font-semibold text-gray-900">Assets</h2>
        
        <AssetSearch
          query={state.searchQuery}
          onSearch={handleSearch}
          resultCount={filteredAssets.length}
        />

        <AssetFilter
          selectedType={state.selectedType}
          onFilterChange={handleFilterChange}
          assetCounts={{
            image: assets.filter((a) => a.asset_type === 'image').length,
            video: assets.filter((a) => a.asset_type === 'video').length,
            audio: assets.filter((a) => a.asset_type === 'audio').length,
            document: assets.filter((a) => a.asset_type === 'document').length,
          }}
        />
      </div>

      {/* Asset Grid */}
      <div className="bg-white rounded-lg shadow-sm overflow-hidden">
        {isLoading && paginatedAssets.length === 0 ? (
          <div className="p-8 text-center">
            <div className="inline-block">
              <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-500" />
            </div>
            <p className="text-gray-600 mt-2">Loading assets...</p>
          </div>
        ) : paginatedAssets.length === 0 ? (
          emptyState || (
            <div className="p-12 text-center">
              <p className="text-gray-500 text-lg">No assets found</p>
              <p className="text-gray-400 text-sm">
                Try adjusting your search or filters
              </p>
            </div>
          )
        ) : (
          <div
            className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-4 p-4"
            role="grid"
            aria-label="Asset gallery"
            aria-rowcount={totalPages}
            aria-colcount={4}
          >
            {paginatedAssets.map((asset, index) => (
              <AssetTile
                key={asset.id}
                asset={asset}
                onClick={() => onAssetSelect?.(asset)}
                tabIndex={0}
                role="gridcell"
                aria-rowindex={Math.floor(index / 4) + 1}
                aria-colindex={(index % 4) + 1}
              />
            ))}
          </div>
        )}
      </div>

      {/* Pagination Controls */}
      {totalPages > 1 && (
        <div className="flex items-center justify-between bg-white rounded-lg shadow-sm p-4">
          <button
            onClick={() => handlePageChange(state.currentPage - 1)}
            disabled={state.currentPage === 1}
            className="px-4 py-2 text-sm font-medium text-gray-700 bg-white border border-gray-300 rounded-md disabled:opacity-50"
            aria-label="Previous page"
          >
            Previous
          </button>

          <div className="text-sm text-gray-600">
            Page {state.currentPage} of {totalPages}
          </div>

          <button
            onClick={() => handlePageChange(state.currentPage + 1)}
            disabled={state.currentPage === totalPages}
            className="px-4 py-2 text-sm font-medium text-gray-700 bg-white border border-gray-300 rounded-md disabled:opacity-50"
            aria-label="Next page"
          >
            Next
          </button>
        </div>
      )}
    </div>
  );
};
